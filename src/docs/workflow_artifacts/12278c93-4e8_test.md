  /**
   * Darwin Ratchet — 达尔文棘轮演化机制
   * 系统只增不减地累积有益特性 (Irreversible Feature Accumulation)
   *
   * API:
   *   Darwin.record(item)  — 记录一次演化 (去重 by id)
   *   Darwin.list()        — 全部特性 (按时间排序)
   *   Darwin.locked()      — 已锁定特性
   *   Darwin.stats()       — 统计
   *   Darwin.onChange(cb)  — 变化订阅
   */
  (function() {
      const STORAGE_KEY = 'poseidonx.darwin.ratchet';
      const listeners = [];
      
      function load() {
          try {
              return JSON.parse(localStorage.getItem(STORAGE_KEY) || '[]');
          } catch (e) {
              return [];
          }
      }
      
      function save(items) {
          localStorage.setItem(STORAGE_KEY, JSON.stringify(items));
          listeners.forEach(cb => { try { cb(items); } catch (e) {} });
          // Broadcast cross-tab
          try {
              const bc = new BroadcastChannel('poseidonx-darwin');
              bc.postMessage({ type: 'update', items });
              bc.close();
          } catch (e) {}
      }
      
      function record(item) {
          if (!item || !item.id) return null;
          const items = load();
          const existing = items.find(x => x.id === item.id);
          const now = new Date().toISOString();
          if (existing) {
              // Update fitness only — core record is locked
              if (item.fitness && item.fitness !== existing.fitness) {
                  existing.fitness = item.fitness;
                  existing.updatedAt = now;
                  save(items);
              }
              return existing;
          }
          const record = {
              id: item.id,
              title: item.title || item.id,
              category: item.category || 'ui',
              description: item.description || '',
              fitness: item.fitness || 'pending',
              lockedAt: item.fitness === 'pass' ? now : null,
              createdAt: now,
              updatedAt: now,
              generation: items.length + 1,
          };
          items.push(record);
          save(items);
          console.log('[Darwin] 🧬 New evolution recorded:', record.title, '(Gen', record.generation + ')');
          return record;
      }
      
      function list() {
          return load().slice().sort((a, b) => (a.createdAt || '').localeCompare(b.createdAt || ''));
      }
      
      function locked() {
          return list().filter(x => x.fitness === 'pass');
      }
      
      function stats() {
          const items = load();
          const byCategory = {};
          items.forEach(i => { byCategory[i.category] = (byCategory[i.category] || 0) + 1; });
          return {
              total: items.length,
              locked: items.filter(i => i.fitness === 'pass').length,
              pending: items.filter(i => i.fitness === 'pending').length,
              rejected: items.filter(i => i.fitness === 'reject').length,
              byCategory,
              lastGeneration: items.length,
          };

          listeners.push(cb);

          try {
              const bc = new BroadcastChannel('poseidonx-darwin');

          } catch (e) {}
          return () => {
              const i = listeners.indexOf(cb);
              if (i >= 0) listeners.splice(i, 1);
          };

      def _should_run(self, agent: ExecAgent, now: datetime) -> bool:

              return (now - last).total_seconds() >= interval
          except (ValueError, TypeError):

          agent.state = ExecState.MONITORING
          agent.metrics.last_cycle_at = now.isoformat()
          cycle_start = time.monotonic()
          result: Dict[str, Any] = {
              "agent": agent.id, "role": agent.role.value, "time": now.isoformat()
          }
  
          try:
              if agent.role == ExecRole.PERCEPTION:
                  result.update(self._run_perception(agent, now, channel_registry))
              elif agent.role == ExecRole.DECISION:
                  result.update(self._run_decision(agent, now, channel_registry))
              elif agent.role == ExecRole.NAVIGATION:
                  result.update(self._run_navigation(agent, now, channel_registry))
              elif agent.role == ExecRole.ENERGY:
                  result.update(self._run_energy(agent, now, channel_registry))
              elif agent.role == ExecRole.FEEDBACK:
                  result.update(self._run_feedback(agent, now))
  
              agent.metrics.cycles_run += 1
              agent.state = ExecState.MONITORING
          except Exception as exc:
              agent.state = ExecState.ERROR
              result["error"] = str(exc)
              logger.error("Exec agent %s cycle failed: %s", agent.id, exc)
  
          elapsed = time.monotonic() - cycle_start
          agent.metrics.uptime_seconds += elapsed
          self.event_log.append(result)
          return result

     * 系统只增不减地累积有益特性 (Irreversible Feature Accumulation)
     *
     * API:
     *   Darwin.record(item)  — 记录一次演化 (去重 by id)
     *   Darwin.list()        — 全部特性 (按时间排序)
     *   Darwin.locked()      — 已锁定特性
     *   Darwin.stats()       — 统计
     *   Darwin.onChange(cb)  — 变化订阅
     */
    (function() {
        const STORAGE_KEY = 'poseidonx.darwin.ratchet';
        const listeners = [];
        
        function load() {
            try {
                return JSON.parse(localStorage.getItem(STORAGE_KEY) || '[]');
            } catch (e) {
                return [];
            }
        }
        
        function save(items) {
            localStorage.setItem(STORAGE_KEY, JSON.stringify(items));
            listeners.forEach(cb => { try { cb(items); } catch (e) {} });
            // Broadcast cross-tab
            try {
                const bc = new BroadcastChannel('poseidonx-darwin');
                bc.postMessage({ type: 'update', items });
                bc.close();
            } catch (e) {}
        }
        
        function record(item) {
            if (!item || !item.id) return null;
            const items = load();
            const existing = items.find(x => x.id === item.id);
            const now = new Date().toISOString();
            if (existing) {
                // Update fitness only — core record is locked
                if (item.fitness && item.fitness !== existing.fitness) {
                    existing.fitness = item.fitness;
                    existing.updatedAt = now;
                    save(items);
                }
                return existing;
            }
            const record = {
                id: item.id,
                title: item.title || item.id,
                category: item.category || 'ui',
                description: item.description || '',
                fitness: item.fitness || 'pending',
                lockedAt: item.fitness === 'pass' ? now : null,
                createdAt: now,
                updatedAt: now,
                generation: items.length + 1,
            };
            items.push(record);
            save(items);
            console.log('[Darwin] 🧬 New evolution recorded:', record.title, '(Gen', record.generation + ')');
            return record;
        }
        
        function list() {
            return load().slice().sort((a, b) => (a.createdAt || '').localeCompare(b.createdAt || ''));
        }
        
        function locked() {
            return list().filter(x => x.fitness === 'pass');
        }
        
        function stats() {
            const items = load();
            const byCategory = {};
            items.forEach(i => { byCategory[i.category] = (byCategory[i.category] || 0) + 1; });
            return {
                total: items.length,
                locked: items.filter(i => i.fitness === 'pass').length,
                pending: items.filter(i => i.fitness === 'pending').length,

    """
    Agent Team API Routes - 双团队管理 REST API
    
    提供构建团队 & 执行团队的状态查询、KPI 考核、

    _execution_team = None

    def set_teams(build_team, execution_team, scheduler, evolution_engine=None):
        """在应用启动时由 main.py 调用，注入团队实例."""
        global _build_team, _execution_team, _scheduler, _evolution_engine

        _execution_team = execution_team
        _scheduler = scheduler
        _evolution_engine = evolution_engine

    # Request / Response Models
    # ---------------------------------------------------------------------------
    
    class TaskAssignment(BaseModel):
        agent_id: str
        task: str

        category: str = "optimization"
        severity: str = "medium"
        title: str
        detail: str

    # Scheduler
    # ---------------------------------------------------------------------------

        return _scheduler.get_status()

    # ---------------------------------------------------------------------------

        ok = _build_team.assign_task(body.agent_id, body.task)
        if not ok:
            raise HTTPException(404, f"Agent '{body.agent_id}' not found")
        return {"status": "assigned", "agent_id": body.agent_id, "task": body.task}
    
    
    @router.get("/build/reports")

        if not _build_team:
            raise HTTPException(503, "Build team not initialized")

        return [r.to_dict() for r in reports]
    
    
    @router.get("/build/issues")
    async def build_issues():
        if not _build_team:
            raise HTTPException(503, "Build team not initialized")
        return _build_team.issue_backlog

    # Execution Team
    # ---------------------------------------------------------------------------
    
    @router.get("/execution/status")

        if not _execution_team:
            raise HTTPException(503, "Execution team not initialized")
        return _execution_team.get_status()
    
    
    @router.get("/execution/agents/{agent_id}")

            raise HTTPException(503, "Execution team not initialized")
        agent = _execution_team.agents.get(agent_id)
        if not agent:
            raise HTTPException(404, f"Agent '{agent_id}' not found")
        return agent.to_dict()
    
    
    @router.get("/execution/reports")

        if not _execution_team:
            raise HTTPException(503, "Execution team not initialized")
        reports = _execution_team.execution_reports[-limit:]
        return [r.to_dict() for r in reports]
    
    
    @router.get("/execution/feedback")
    async def execution_feedback():
        if not _execution_team:
            raise HTTPException(503, "Execution team not initialized")
        return [item.to_dict() for item in _execution_team.feedback_queue]
    
    
    @router.post("/execution/feedback")

        if not _execution_team:
            raise HTTPException(503, "Execution team not initialized")
        item = _execution_team.submit_feedback(
            category=body.category,

            title=body.title,
            detail=body.detail,
        )
        return item.to_dict()
    
    
    # ---------------------------------------------------------------------------
    # Combined
    # ---------------------------------------------------------------------------
    
    @router.get("/overview")
    async def teams_overview():
        """一站式获取双团队全局概览."""
        result: Dict[str, Any] = {}
        if _build_team:
            bs = _build_team.get_status()
            result["build_team"] = {
                "health": bs["health"],
                "agent_count": bs["agent_count"],
                "metrics": bs["metrics"],
            }
        if _execution_team:
            es = _execution_team.get_status()
            result["execution_team"] = {
                "health": es["health"],
                "agent_count": es["agent_count"],
                "metrics": es["metrics"],
            }
        if _scheduler:
            result["scheduler"] = _scheduler.get_status()
        if _evolution_engine:
            result["evolution"] = _evolution_engine.get_status()
        return result
    
    
    # ---------------------------------------------------------------------------

    @router.get("/evolution/status")

        """获取自我演进引擎状态。"""

     * 系统只增不减地累积有益特性 (Irreversible Feature Accumulation)
     *
     * API:
     *   Darwin.record(item)  — 记录一次演化 (去重 by id)

     *   Darwin.onChange(cb)  — 变化订阅

    (function() {
        const STORAGE_KEY = 'poseidonx.darwin.ratchet';
        const listeners = [];

            try {
                return JSON.parse(localStorage.getItem(STORAGE_KEY) || '[]');
            } catch (e) {
                return [];
            }
        }

            localStorage.setItem(STORAGE_KEY, JSON.stringify(items));
            listeners.forEach(cb => { try { cb(items); } catch (e) {} });

                bc.close();
            } catch (e) {}

            if (!item || !item.id) return null;
            const items = load();
            const existing = items.find(x => x.id === item.id);
            const now = new Date().toISOString();
            if (existing) {
                // Update fitness only — core record is locked
                if (item.fitness && item.fitness !== existing.fitness) {
                    existing.fitness = item.fitness;
                    existing.updatedAt = now;
                    save(items);
                }
                return existing;

                lockedAt: item.fitness === 'pass' ? now : null,

            listeners.forEach(cb => { try { cb(items); } catch (e) {} });
            // Broadcast cross-tab

                bc.postMessage({ type: 'update', items });
                bc.close();

            if (!item || !item.id) return null;
            const items = load();
            const existing = items.find(x => x.id === item.id);
            const now = new Date().toISOString();
            if (existing) {
                // Update fitness only — core record is locked
                if (item.fitness && item.fitness !== existing.fitness) {
                    existing.fitness = item.fitness;
                    existing.updatedAt = now;
                    save(items);
                }
                return existing;
            }
            const record = {
                id: item.id,
                title: item.title || item.id,
                category: item.category || 'ui',
                description: item.description || '',
                fitness: item.fitness || 'pending',
                lockedAt: item.fitness === 'pass' ? now : null,
                createdAt: now,
                updatedAt: now,
                generation: items.length + 1,
            };
            items.push(record);
            save(items);
            console.log('[Darwin] 🧬 New evolution recorded:', record.title, '(Gen', record.generation + ')');
            return record;

            return load().slice().sort((a, b) => (a.createdAt || '').localeCompare(b.createdAt || ''));

            return list().filter(x => x.fitness === 'pass');

                return (now - last).total_seconds() >= interval
            except (ValueError, TypeError):

        def _execute_agent_cycle(
            self, agent: ExecAgent, now: datetime,
            channel_registry: Optional[Dict] = None,
        ) -> Dict[str, Any]:
            agent.state = ExecState.MONITORING
            agent.metrics.last_cycle_at = now.isoformat()
            cycle_start = time.monotonic()
            result: Dict[str, Any] = {
                "agent": agent.id, "role": agent.role.value, "time": now.isoformat()
            }
    
            try:
                if agent.role == ExecRole.PERCEPTION:
                    result.update(self._run_perception(agent, now, channel_registry))
                elif agent.role == ExecRole.DECISION:
                    result.update(self._run_decision(agent, now, channel_registry))
                elif agent.role == ExecRole.NAVIGATION:
                    result.update(self._run_navigation(agent, now, channel_registry))
                elif agent.role == ExecRole.ENERGY:
                    result.update(self._run_energy(agent, now, channel_registry))
                elif agent.role == ExecRole.FEEDBACK:
                    result.update(self._run_feedback(agent, now))
    
                agent.metrics.cycles_run += 1
                agent.state = ExecState.MONITORING
            except Exception as exc:
                agent.state = ExecState.ERROR
                result["error"] = str(exc)
                logger.error("Exec agent %s cycle failed: %s", agent.id, exc)
    
            elapsed = time.monotonic() - cycle_start
            agent.metrics.uptime_seconds += elapsed
            self.event_log.append(result)
            return result

#### **1. 执行摘要**

3.  **任务分配不明确:** PM分解中，任务被分配给`build_pm`，但未进一步分解给具体的开发角色（如前端3D工程师），可能导致执行混乱。

**建议: 在进入编码阶段前，必须由架构师补充详细的技术实现方案，并由PM重新细化任务分配。**

#### **2. 详细审查结果**

##### **2.1. 任务理解与分解 (PM 产出审查)**

*   **任务:** `让货船以双体船为圆心动起来`
*   **PM 分解:** 将任务分配给 `build_pm`，并输出了一份任务分解文档。
*   **审查发现:**
    *   **问题 1 (严重): 任务粒度太粗。** PM的分解停留在“分配任务”层面，没有将“让货船动起来”这个具体开发任务拆解为可执行的技术子任务（例如：`[前端3D] 定位双体船和货船模型`、`[前端3D] 实现公转动画逻辑`、`[前端3D] 集成到主循环`）。这会导致开发人员（如`build_frontend_engineer`）接到任务后仍然不知道具体要做什么。
    *   **问题 2 (中等): 角色错位。** 任务要求“给build团队的PM分配一个任务”，但PM的分解产出仍然是给PM自己的。正确的流程应该是PM将任务分解后，分配给具体的开发角色（如`build_3d_engineer`或`build_frontend_engineer`）。
    *   **修复建议:**
        1.  PM应重新分解任务，产出类似以下的技术任务清单：
            *   `T-001`: 在`digital-twin.html`或相关3D场景入口文件中，定位并获取“双体船”和“货船”的Three.js对象引用。
            *   `T-002`: 实现一个`animateCargoShip()`函数，该函数在每一帧更新货船的位置，使其围绕双体船的中心点做圆周运动。
            *   `T-003`: 将`animateCargoShip()`函数集成到场景的主渲染循环（`requestAnimationFrame`）中。
            *   `T-004`: 添加一个简单的控制参数（如旋转半径、速度），方便后续调整。
        2.  将上述任务分配给`build_frontend_engineer`或`build_3d_engineer`。

##### **2.2. 技术调研 (Researcher 产出审查)**

*   **产出:** 一份关于Three.js `Object3D`、`lookAt`、`requestAnimationFrame`等技术的研究文档。
*   **审查发现:**
    *   **问题 3 (严重): 技术方案不完整。** 调研提到了`Object3D`和`lookAt`，但`lookAt`是让物体朝向另一个点，**无法实现“围绕某点公转”**。实现公转的核心是**更新物体的位置**，通常使用三角函数（`Math.sin`, `Math.cos`）计算圆周上的坐标，或者使用`Object3D`的父子层级关系（将货船作为双体船的子对象，然后旋转双体船）。
    *   **问题 4 (中等): 未考虑项目现有架构。** 调研没有提及项目现有的3D场景是如何组织的。例如，模型是加载的GLB文件（`GLB_20251223141542.glb`）还是代码生成的？场景的入口是哪个HTML文件？主循环在哪里？这些信息对于实现至关重要。
    *   **修复建议:**
        1.  **修正技术方案:** 明确使用**更新位置**的方法来实现公转。例如，在动画循环中，根据当前角度计算货船的新坐标：
            ```javascript
            // 伪代码
            const radius = 50; // 公转半径
            const speed = 0.01; // 公转速度
            let angle = 0;

            function animate() {
                angle += speed;
                cargoShip.position.x = catamaran.position.x + radius * Math.cos(angle);
                cargoShip.position.z = catamaran.position.z + radius * Math.sin(angle);
                // 可选：让货船始终面向双体船
                cargoShip.lookAt(catamaran.position);
                requestAnimationFrame(animate);
            }
            ```
        2.  **补充上下文调研:** 研究员需要明确指出：
            *   哪个HTML文件是3D场景的入口（很可能是`digital-twin.html`或`index.html`）。
            *   主渲染循环在哪个JS文件中（很可能是`main.js`或`demo.js`）。
            *   双体船和货船在代码中的变量名是什么。

##### **2.3. 架构设计 (Architect 产出审查)**

*   **产出:** 一份架构设计文档，描述了模块划分、数据流和接口。
*   **审查发现:**
    *   **问题 5 (严重): 设计过于抽象，缺乏实现细节。** 架构文档描述了“前端3D引擎”和“动画控制模块”，但没有定义：
        *   **接口:** `animateCargoShip(catamaranObj, cargoShipObj, config)` 这个函数的输入参数和输出是什么？
        *   **数据流:** 动画参数（半径、速度）从哪里来？是硬编码、来自配置文件还是通过API动态下发？
        *   **模块交互:** “动画控制模块”如何与现有的“场景管理模块”或“主循环”交互？
    *   **问题 6 (中等): 未考虑异常情况。** 如果双体船或货船的模型加载失败，或者对象引用为`null`，动画逻辑应该如何处理？设计文档中没有提及任何错误处理或降级方案。
    *   **修复建议:**
        1.  **细化接口设计:** 架构师应提供明确的接口定义，例如：
            ```typescript
            // 定义在某个模块中
            interface CargoShipOrbitConfig {
                radius: number;
                speed: number; // 弧度/帧
                startAngle?: number;
            }

            interface IOrbitController {
                // 初始化，传入两个3D对象
                init(catamaran: THREE.Object3D, cargoShip: THREE.Object3D, config: CargoShipOrbitConfig): void;
                // 每帧调用，更新货船位置
                update(deltaTime: number): void;
                // 销毁，清理资源
                dispose(): void;
            }
            ```
        2.  **增加健壮性设计:** 在`init`方法中增加参数校验，如果传入的对象无效，应抛出明确的错误或返回`false`。在`update`方法中，应检查对象是否仍然存在于场景中。

#### **3. 风险评估**

| 风险类别 | 风险描述 | 可能性 | 影响 | 缓解措施 |
| :--- | :--- | :--- | :--- | :--- |
| **技术实现** | 开发人员可能错误地使用`lookAt`或`Object3D`的`rotate`方法来实现公转，导致动画效果错误。 | 高 | 高 | 在架构设计中明确使用“更新位置”方案，并在代码审查时重点检查。 |
| **任务执行** | 由于任务分解不细，开发人员可能花费大量时间理解需求，或实现与预期不符的功能。 | 高 | 中 | PM需重新分解任务，并组织一次简短的会议对齐需求。 |
| **集成风险** | 新添加的动画逻辑可能与现有的场景控制（如相机控制、天气效果）发生冲突。 | 中 | 高 | 架构师需定义清晰的模块边界，确保新模块不干扰现有功能。QA需进行集成测试。 |
| **性能风险** | 如果公转计算或模型更新效率低下，可能导致帧率下降。 | 低 | 中 | 开发时使用`deltaTime`而非固定增量，避免在大量对象上执行复杂计算。 |

#### **4. 测试建议 (为后续阶段准备)**

一旦代码实现完成，QA将执行以下测试：

1.  **功能测试:**
    *   **TC-001:** 验证货船是否围绕双体船做圆周运动。
    *   **TC-002:** 验证货船的运动轨迹是否为圆形，而非椭圆或不规则形状。
    *   **TC-003:** 验证货船在运动过程中是否始终面向双体船（如果设计如此）。
    *   **TC-004:** 验证修改配置参数（半径、速度）后，动画效果是否随之改变。
2.  **边界与异常测试:**
    *   **TC-005:** 验证当双体船移动时，货船是否跟随其作为新的圆心继续公转。
    *   **TC-006:** 验证当货船或双体船模型被隐藏或移除时，动画逻辑是否优雅地停止或报错，而不会导致页面崩溃。
    *   **TC-007:** 验证在极低帧率下，动画是否依然平滑（使用`deltaTime`）。
3.  **集成测试:**
    *   **TC-008:** 验证在切换场景主题、开启/关闭天气效果等操作下，公转动画是否不受影响。
    *   **TC-009:** 验证在`digital-twin.html`页面加载时，动画是否自动启动。
