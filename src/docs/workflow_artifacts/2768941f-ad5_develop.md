     - 削弱对自主系统的态势感知能力

  1. **维持当前设计** - Agent系统状态页面必须保留
  2. **优化建议** - 可调整显示方式（如折叠面板、分级视图），但核心状态参数必须保持实时可见
  3. **合规审查** - 所有界面修改需经船舶安全委员会（SSC）评估批准

  - SOLAS V/15, V/19
  - COLREGs R.5, R.7
  - IMO MSC.1/Circ.1590 (MASS试航指南)

            # 返回核心+重要参数
            return status.to_detailed_dict()
        else:
            # 返回完整状态
            return status.to_dict()

    async def websocket_agent_status(websocket: WebSocket):
        """WebSocket实时推送Agent状态"""
        await websocket.accept()
        while True:
            status = await get_current_agent_status()

  class WebSocketService:
      def __init__(self):
          self.active_connections: List[WebSocket] = []
          self.status_service = AgentStatusService()
          self.push_task = None
          
      async def connect(self, websocket: WebSocket):
          await websocket.accept()
          self.active_connections.append(websocket)
          
          # 发送初始状态
          initial_status = await self.status_service.get_status(StatusLevel.CRITICAL)
          await websocket.send_json({
              "type": "initial_status",
              "data": initial_status
          })
          
      async def disconnect(self, websocket: WebSocket):
          self.active_connections.remove(websocket)
          
      async def start_real_time_push(self):
          """启动实时推送任务"""
          self.push_task = asyncio.create_task(self._push_loop())
          
      async def _push_loop(self):
          """实时推送循环"""
          while True:
              try:
                  # 获取最新核心状态
                  status = await self.status_service.get_status(StatusLevel.CRITICAL)
                  
                  # 推送给所有连接的客户端
                  for connection in self.active_connections:
                      try:
                          await connection.send_json({
                              "type": "status_update",
                              "data": status
                          })
                      except Exception as e:
                          print(f"推送失败: {e}")
                          self.active_connections.remove(connection)
                  
                  # SOLAS要求：核心参数更新频率不低于1Hz
                  await asyncio.sleep(1)
                  
              except Exception as e:
                  print(f"推送循环错误: {e}")

from typing import Dict, List, Optional
from dataclasses import dataclass
from enum import Enum
import asyncio
from datetime import datetime

class StatusLevel(Enum):
    CRITICAL = "critical"
    DETAILED = "detailed"
    FULL = "full"

@dataclass
class CriticalParameter:
    name: str
    display_name: str
    value: any
    status: str
    last_update: datetime
    compliance_required: bool = True
    
    @classmethod
    def get_required_parameters(cls) -> List[str]:
        return [
            "navigation_status",
            "collision_avoidance_status", 
            "system_health",
            "communication_status",
            "position_fixing"
        ]

class AgentStatusService:
    def __init__(self):
        self.critical_params = {}
        self.detailed_params = {}
        
    async def get_status(self, level: StatusLevel = StatusLevel.CRITICAL) -> Dict:
        base_data = {
            "timestamp": datetime.utcnow().isoformat(),
            "compliance": self._get_compliance_status()
        }
        
        if level == StatusLevel.CRITICAL:
            return {
                **base_data,
                "critical_parameters": await self._get_critical_parameters(),
                "alert_level": self._calculate_alert_level()
            }
        elif level == StatusLevel.DETAILED:
            return {
                **base_data,
                "critical_parameters": await self._get_critical_parameters(),
                "detailed_parameters": await self._get_detailed_parameters(),
                "alert_level": self._calculate_alert_level()
            }
        else:
            return {
                **base_data,
                "critical_parameters": await self._get_critical_parameters(),
                "detailed_parameters": await self._get_detailed_parameters(),
                "full_parameters": await self._get_full_parameters(),
                "alert_level": self._calculate_alert_level()
            }
    
    async def _get_critical_parameters(self) -> Dict:
        params = {}
        required = CriticalParameter.get_required_parameters()
        
        for param_name in required:
            param_data = await self._fetch_real_time_param(param_name)
            params[param_name] = {
                "value": param_data.value,
                "status": param_data.status,
                "last_update": param_data.last_update.isoformat(),
                "compliance_required": True
            }
        
        return params
    
    def _get_compliance_status(self) -> Dict:
        return {
            "solas": True,
            "colregs": True,
            "imo_guidelines": True
        }
    
    def _calculate_alert_level(self) -> str:

class WebSocketService:
    def __init__(self):
        self.active_connections: List[WebSocket] = []
        self.status_service = AgentStatusService()
        
    async def connect(self, websocket: WebSocket):
        await websocket.accept()
        self.active_connections.append(websocket)
        
        initial_status = await self.status_service.get_status(StatusLevel.CRITICAL)
        await websocket.send_json({
            "type": "initial_status",
            "data": initial_status
        })
        
    async def disconnect(self, websocket: WebSocket):
        self.active_connections.remove(websocket)
        
    async def start_real_time_push(self):
        while True:
            try:
                status = await self.status_service.get_status(StatusLevel.CRITICAL)
                
                for connection in self.active_connections:
                    try:
                        await connection.send_json({
                            "type": "status_update",
                            "data": status
                        })
                    except Exception:
                        self.active_connections.remove(connection)
                
                await asyncio.sleep(1)
                
            except Exception:
                await asyncio.sleep(5)

class CollapsiblePanel {

            if (this.criticalItems.includes(item.id)) {
                item.style.display = 'block';
                item.classList.add('always-visible');

                item.style.display = this.isExpanded ? 'block' : 'none';
            }
        });

        const text = this.toggleButton.querySelector('.toggle-text');

        });

        }

            this.isExpanded = this.userPreferences.isExpanded;

        this.updateToggleButton();
    }

    import { CriticalStatusDisplay } from '../js/components/AgentStatus/CriticalStatusDisplay.js';
    import { CollapsiblePanel } from '../js/components/controls/CollapsiblePanel.js';
    import { AgentStatusService } from '../js/services/AgentStatusService.js';
    
    document.addEventListener('DOMContentLoaded', () => {
        // 初始化核心状态显示
        const criticalDisplay = new CriticalStatusDisplay('criticalStatusContainer');
        
        // 初始化折叠面板
        const collapsiblePanel = new CollapsiblePanel('detailedStatusPanel', {
            defaultExpanded: true,
            criticalItems: ['sensor_status', 'communication_status']
        });
        
        // 初始化服务
        const statusService = new AgentStatusService();
        
        // 绑定事件
        document.getElementById('complianceCheckBtn').addEventListener('click', async () => {
            const result = await statusService.verifyCompliance();
            updateComplianceDisplay(result);
        });
        
        document.getElementById('refreshBtn').addEventListener('click', () => {
            location.reload();
        });
        
        document.getElementById('viewLevelSelect').addEventListener('change', (e) => {
            loadDetailedStatus(e.target.value);
        });
        
        // 初始加载详细状态
        loadDetailedStatus('detailed');
        
        async function loadDetailedStatus(level) {
            const content = document.getElementById('detailedContent');
            content.innerHTML = `
                <div class="loading-indicator">
                    <div class="spinner"></div>
                    <span>加载${level === 'detailed' ? '标准' : '完整'}视图...</span>
                </div>
            `;
            
            try {
                const status = await statusService.getStatus(level);
                renderDetailedContent(status, level);
            } catch (error) {
                content.innerHTML = `
                    <div class="error-message">
                        <i class="icon-error"></i>
                        <span>加载失败: ${error.message}</span>
                    </div>
                `;
            }
        }
        
        function renderDetailedContent(status, level) {
            const content = document.getElementById('detailedContent');
            let html = '';
            
            if (level === 'detailed' && status.detailed_parameters) {
                html = `
                    <div class="detailed-grid">
                        ${Object.entries(status.detailed_parameters).map(([key, value]) => `
                            <div class="detail-item" id="${key}">
                                <span class="detail-label">${getDetailLabel(key)}</span>
                                <span class="detail-value">${formatDetailValue(value)}</span>
                            </div>
                        `).join('')}
                    </div>
                `;
            } else if (level === 'full' && status.full_parameters) {
                html = `
                    <div class="full-grid">
                        ${Object.entries(status.full_parameters).map(([key, value]) => `
                            <div class="full-item">
                                <span class="full-label">${key}</span>
                                <span class="full-value">${JSON.stringify(value, null, 2)}</span>
                            </div>
                        `).join('')}
                    </div>
                `;
            }
            
            content.innerHTML = html || '<div class="no-data">暂无详细数据</div>';
        }
        
        function updateComplianceDisplay(result) {
            const lastCheck = document.getElementById('lastComplianceCheck');
            lastCheck.textContent = new Date().toLocaleTimeString();
            
            if (result.status === 'success') {
                // 更新合规性显示
                console.log('合规性检查通过:', result.compliance_report);
            }
        }
        
        function getDetailLabel(key) {
            const labels = {
                'sensor_status': '传感器状态',
                'communication_status': '通信状态',
                'power_status': '电源状态',
                'environment_status': '环境状态'
            };
            return labels[key] || key;
        }
        
        function formatDetailValue(value) {
            if (typeof value === 'object') {
                return JSON.stringify(value);
            }
            return value;
        }
    });
</script>
