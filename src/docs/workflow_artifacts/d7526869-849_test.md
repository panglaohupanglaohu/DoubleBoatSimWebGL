#!/usr/bin/env python3
"""

import os
import re

def check_file_consistency():
    """检查文件修改的一致性"""
    print("🔍 检查文件修改一致性")
    
    issues = []
    
    # 1. 检查所有相关文件是否存在
    required_files = [
        "src/frontend/device_health.html",
        "src/frontend/js/menu.js", 
        "src/frontend/js/app.js",
        "src/backend/api/device_health.py"
    ]
    
    for file_path in required_files:
        if not os.path.exists(file_path):
            issues.append(f"文件不存在: {file_path}")
    
    if issues:
        return False, issues
    
    # 2. 检查HTML文件
    with open("src/frontend/device_health.html", 'r', encoding='utf-8') as f:
        html_content = f.read()
    
    # 检查菜单项数量
    menu_items = re.findall(r'<li><a href="#[^"]+">[^<]+</a></li>', html_content)
    if len(menu_items) != 4:  # 应该只有4个菜单项（移除注册表后）
        issues.append(f"HTML菜单项数量不正确: 期望4个，实际{len(menu_items)}个")
    
    # 3. 检查JS文件一致性
    with open("src/frontend/js/menu.js", 'r', encoding='utf-8') as f:
        menu_js = f.read()
    
    # 检查menuItems数组
    menu_items_match = re.search(r'const menuItems = \[(.*?)\];', menu_js, re.DOTALL)
    if menu_items_match:
        menu_items_str = menu_items_match.group(1)
        if "'registry'" in menu_items_str:
            issues.append("menu.js 中仍然包含 'registry' 菜单项")
    
    # 4. 检查功能开关
    with open("src/frontend/js/app.js", 'r', encoding='utf-8') as f:
        app_js = f.read()
    
    if "registryEnabled: true" in app_js:
        issues.append("app.js 中 registryEnabled 仍然为 true")
    
    return len(issues) == 0, issues

def check_page_functionality():

        html_content = f.read()

    required_scripts = ["js/menu.js", "js/app.js"]
    for script in required_scripts:
        if script not in html_content:
            issues.append(f"HTML未加载必要脚本: {script}")
    
    # 检查CSS加载
    if 'css/styles.css' not in html_content:
        issues.append("HTML未加载CSS文件")
    
    return len(issues) == 0, issues

def check_backend_compatibility():
    """检查后端兼容性"""
    print("🔍 检查后端兼容性")
    
    issues = []
    
    with open("src/backend/api/device_health.py", 'r', encoding='utf-8') as f:
        backend_content = f.read()
    
    # 检查必要的API端点是否存在
    required_apis = [
        "@router.get(\"/overview\")",
        "@router.get(\"/metrics\")",
        "@router.get(\"/logs\")",
        "@router.get(\"/settings\")"
    ]
    
    for api in required_apis:
        if api not in backend_content:
            issues.append(f"后端缺少必要API: {api}")
    
    # 检查注册表API是否有适当注释（如果保留）
    if "@router.get(\"/registry\")" in backend_content:
        # 检查是否有注释说明
        lines = backend_content.split('\n')
        for i, line in enumerate(lines):
            if "@router.get(\"/registry\")" in line:
                # 检查前几行是否有注释
                has_comment = False
                for j in range(max(0, i-3), i):
                    if '#' in lines[j] and ('隐藏' in lines[j] or '内部' in lines[j] or '保留' in lines[j]):
                        has_comment = True
                        break
                
                if not has_comment:
                    issues.append("注册表API缺少适当的注释说明")
    
    return len(issues) == 0, issues
